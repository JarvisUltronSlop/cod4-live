cmake_minimum_required(VERSION 3.25)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" COD4LIVE_VERSION LIMIT_COUNT 1)
project(cod4-live VERSION ${COD4LIVE_VERSION} LANGUAGES C CXX RC)
set(version_patch 0)
if(PROJECT_VERSION_PATCH)
  set(version_patch ${PROJECT_VERSION_PATCH})
endif()
set(COD4LIVE_VERSION_COMMAS "${PROJECT_VERSION_MAJOR},${PROJECT_VERSION_MINOR},${version_patch},0")
configure_file(native/release_version.h.in "${CMAKE_BINARY_DIR}/generated/release_version.h" @ONLY)
function(cod4live_version_resource target description filename)
  set(COD4LIVE_FILE_DESCRIPTION "${description}")
  set(COD4LIVE_ORIGINAL_FILENAME "${filename}")
  configure_file(native/version.rc.in "${CMAKE_BINARY_DIR}/generated/${target}-version.rc" @ONLY)
  target_sources(${target} PRIVATE "${CMAKE_BINARY_DIR}/generated/${target}-version.rc")
endfunction()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(CTest)
option(COD4LIVE_BUILD_ENGINE "Build client and dedicated server" ON)
set(COD4LIVE_INSTALLER_PAYLOAD "" CACHE FILEPATH "Installer CAB payload")
set(COD4LIVE_INSTALLER_TERMS "${CMAKE_CURRENT_SOURCE_DIR}/.cache/dxsdk/LICENSE.txt" CACHE FILEPATH "Microsoft D3DX runtime license shown before installation")
if(COD4LIVE_BUILD_ENGINE)
  if(NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "The classic engine requires -A Win32")
  endif()
  add_subdirectory(engine)
  set_target_properties(KisakCOD-mp PROPERTIES OUTPUT_NAME "cod4-live")
  set_target_properties(KisakCOD-dedi PROPERTIES OUTPUT_NAME "cod4-live_server")
  foreach(target KisakCOD-mp KisakCOD-dedi)
    set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "$<1:${CMAKE_SOURCE_DIR}/runtime>")
    target_compile_definitions(${target} PRIVATE COD4LIVE=1)
    # Keep optimized-build symbols beside the binaries for crash diagnostics;
    # PDBs remain excluded from installer payloads and source control.
    target_link_options(${target} PRIVATE "$<$<CONFIG:Release>:/DEBUG:FULL>" "$<$<CONFIG:Release>:/OPT:REF>" "$<$<CONFIG:Release>:/OPT:ICF>")
  endforeach()
  cod4live_version_resource(KisakCOD-mp "cod4-live multiplayer client" "cod4-live.exe")
  cod4live_version_resource(KisakCOD-dedi "cod4-live dedicated server" "cod4-live_server.exe")
  include(engine/src/live/controller.cmake OPTIONAL)
  include(engine/src/live/bot_warfare.cmake)
endif()
if(COD4LIVE_INSTALLER_PAYLOAD)
  if(NOT EXISTS "${COD4LIVE_INSTALLER_TERMS}")
    message(FATAL_ERROR "Prepare the D3DX runtime license before building setup")
  endif()
  configure_file(native/setup-payload.rc.in "${CMAKE_BINARY_DIR}/setup-payload.rc" @ONLY)
  add_executable(cod4_live_setup WIN32 native/setup.cpp native/install_package.cpp native/desktop.rc "${CMAKE_BINARY_DIR}/setup-payload.rc")
  target_compile_options(cod4_live_setup PRIVATE /utf-8)
  target_include_directories(cod4_live_setup PRIVATE "${CMAKE_BINARY_DIR}/generated")
  target_compile_definitions(cod4_live_setup PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(cod4_live_setup PRIVATE comctl32 shell32 ole32 cabinet bcrypt dwmapi uxtheme)
  target_link_options(cod4_live_setup PRIVATE /MANIFEST:NO)
  set_target_properties(cod4_live_setup PROPERTIES OUTPUT_NAME "cod4-live_setup")
  cod4live_version_resource(cod4_live_setup "cod4-live Windows installer" "cod4-live_setup.exe")
endif()
if(BUILD_TESTING AND EXISTS "${CMAKE_SOURCE_DIR}/tests/CMakeLists.txt")
  add_subdirectory(tests)
endif()
if(BUILD_TESTING AND COD4LIVE_INSTALLER_PAYLOAD)
  add_executable(install_package_tests tests/install_package_tests.cpp native/install_package.cpp "${CMAKE_BINARY_DIR}/setup-payload.rc")
  target_include_directories(install_package_tests PRIVATE native)
  target_compile_definitions(install_package_tests PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_compile_options(install_package_tests PRIVATE /UNDEBUG /utf-8)
  target_link_libraries(install_package_tests PRIVATE shell32 ole32 cabinet bcrypt dwmapi uxtheme comctl32)
  add_test(NAME install_package COMMAND install_package_tests)
  # Explicit real-data acceptance only; never runs from the automated fixture suite.
  add_executable(install_acceptance tests/install_acceptance.cpp native/install_package.cpp "${CMAKE_BINARY_DIR}/setup-payload.rc")
  target_include_directories(install_acceptance PRIVATE native)
  target_compile_definitions(install_acceptance PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_compile_options(install_acceptance PRIVATE /utf-8)
  target_link_libraries(install_acceptance PRIVATE shell32 ole32 cabinet bcrypt dwmapi uxtheme comctl32)
endif()
